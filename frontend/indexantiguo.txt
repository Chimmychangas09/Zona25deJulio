        function initializeGeolocation() {
            const box = document.getElementById('geoStatusBox');
            const title = document.getElementById('geoTitle');
            const coordsSpan = document.getElementById('geoCoords');
            const btnSubmit = document.getElementById('btnReportarSubmit');

            if (!navigator.geolocation) {
                box.className = 'geo-box error';
                title.innerText = 'Sensor Incompatible';
                coordsSpan.innerText = 'Tu navegador no soporta la API de Geolocalización de HTML5.';
                return;
            }

            box.className = 'geo-box searching';
            title.innerText = 'Adquiriendo Coordenadas GPS...';
            coordsSpan.innerText = 'Buscando satélites de posicionamiento en El Alto...';
            btnSubmit.disabled = true;

            navigator.geolocation.getCurrentPosition(
                (position) => {
                    const lat = position.coords.latitude;
                    const lng = position.coords.longitude;
                    
                    document.getElementById('incLatitud').value = lat;
                    document.getElementById('incLongitud').value = lng;

                    box.className = 'geo-box';
                    title.innerText = 'Ubicación Satelital Fijada';
                    coordsSpan.innerText = `Latitud: ${lat.toFixed(6)} | Longitud: ${lng.toFixed(6)} (Precisión: ${position.coords.accuracy.toFixed(1)}m)`;
                    btnSubmit.disabled = false;
                },
                (error) => {
                    box.className = 'geo-box error';
                    title.innerText = 'Fallo del Sensor GPS';
                    btnSubmit.disabled = true;
                    switch(error.code) {
                        case error.PERMISSION_DENIED:
                            coordsSpan.innerText = "Acceso denegado. Debes otorgar permisos de GPS para reportar.";
                            break;
                        default:
                            coordsSpan.innerText = "No se logró capturar la ubicación física actual.";
                    }
                },
                { enableHighAccuracy: true, timeout: 10000 }
            );
        }

let BACKEND_INCIDENCIAS_CACHE = [];

async function loadAdminDashboard() {
    try {
        const token = localStorage.getItem('token');
        
        // Petición 1: Incidencias y clasificación
        const resInc = await fetch('http://localhost:8000/api/incidencias', {
            headers: { 'Authorization': token }
        });
        const dataInc = await resInc.json();
        
        if (dataInc.status === 'success') {
            // 🚀 INTERCEPCIÓN MAESTRA: Guardamos los datos puros del backend en nuestra caché global
            BACKEND_INCIDENCIAS_CACHE = dataInc.data;
            
            // Renderizamos inicialmente con la lista completa de la caché
            renderMetricsAndIncidencias(BACKEND_INCIDENCIAS_CACHE);
            renderSectorsDensity(BACKEND_INCIDENCIAS_CACHE); // Algoritmo US06
        }

        // Petición 2: Bitácora de Auditoría en Vivo
        const resAudit = await fetch('http://localhost:8000/api/admin/auditoria', {
            headers: { 'Authorization': token }
        });
        const dataAudit = await resAudit.json();
        
        if (dataAudit.status === 'success') {
            renderAuditTimeline(dataAudit.data); // Renderizado US04
        }
    } catch (error) {
        console.error("Error en el flujo del Bloque Upgrade:", error);
    }
}

function renderMetricsAndIncidencias(incidencias) {
    let pendientes = 0, proceso = 0, resueltas = 0;
    const container = document.getElementById('adminIncidenciasList');
    container.innerHTML = '';

    incidencias.forEach(item => {
        if (item.estado === 'Pendiente') pendientes++;
        if (item.estado === 'En Proceso') proceso++;
        if (item.estado === 'Resuelto') resueltas++;

        const badgeClass = item.estado === 'Pendiente' ? 'badge-danger' : (item.estado === 'En Proceso' ? 'badge-warning' : 'badge-success');

        // 🖼️ TRATAMIENTO DE IMÁGENES REPARADO CON FOTO_URL
        let contenedorImagenHTML = "";
        
        // 1. Usamos la propiedad correcta 'foto_url' y limpiamos los caracteres de escape (\/)
        if (item.foto_url && item.foto_url.trim() !== "") {
            const rutaLimpia = item.foto_url.replace(/\\/g, ''); // Quita las barras invertidas intrusas
            
            // 2. Construimos la URL real apuntando al backend (puerto 8000)
            const URL_FOTO = `http://localhost:8000/${rutaLimpia}`;
            
            // Salvavidas onerror por si el archivo físico fue borrado en la carpeta uploads
            contenedorImagenHTML = `
                <img src="${URL_FOTO}" alt="Evidencia" onerror="this.style.display='none'; this.nextElementSibling.style.display='flex';">
                <div class="fallback-foto-box" style="display:none; width:140px; height:100%; background:#f2f4f4; align-items:center; justify-content:center; text-align:center; color:#7f8c8d; font-size:0.8rem; padding:10px; flex-direction:column;">
                    <i class="ti ti-photo-off" style="font-size:1.5rem;"></i><span>No disponible</span>
                </div>
            `;
        } else {
            // Si el campo directamente venía vacío en la base de datos
            contenedorImagenHTML = `
                <div class="fallback-foto-box" style="display:flex; width:140px; height:100%; background:#eeeeee; align-items:center; justify-content:center; text-align:center; color:#7f8c8d; font-size:0.8rem; padding:10px; flex-direction:column; gap:5px;">
                    <i class="ti ti-camera-off" style="font-size:1.5rem;"></i>
                    <span>Sin Registro</span>
                </div>
            `;
        }

        container.innerHTML += `
            <div class="admin-item-card border-${item.estado.toLowerCase().replace(' ', '')}">
                <div class="admin-card-img" style="width:140px; min-width:140px; background:#f8f9fa; display:flex; align-items:center; justify-content:center; overflow:hidden;">
                    ${contenedorImagenHTML}
                </div>
                <div class="admin-card-body">
                    <span class="badge ${badgeClass}">${item.estado}</span>
                    <h4>${item.tipo_danio}</h4>
                    <p class="sector-text"><i class="ti ti-map-pin"></i> ${item.sector || 'Sector General'}</p>
                    <p class="desc-text">${item.descripcion}</p>
                    <div class="control-actions">
                        <label>Cambiar Estado:</label>
                        <select onchange="updateIncidenciaEstado(${item.id}, this.value)">
                            <option value="Pendiente" ${item.estado === 'Pendiente' ? 'selected' : ''}>Pendiente 🔴</option>
                            <option value="En Proceso" ${item.estado === 'En Proceso' ? 'selected' : ''}>En Proceso 🟡</option>
                            <option value="Resuelto" ${item.estado === 'Resuelto' ? 'selected' : ''}>Resuelto 🟢</option>
                        </select>
                    </div>
                </div>
            </div>
        `;
    });

    document.getElementById('countPendientes').innerText = pendientes;
    document.getElementById('countProceso').innerText = proceso;
    document.getElementById('countResueltas').innerText = resueltas;
}


// Algoritmo de ordenamiento volumétrico por densidad de daños (US06)
function renderSectorsDensity(incidencias) {
    const sectorCounts = {};
    incidencias.forEach(item => {
        sectorCounts[item.sector] = (sectorCounts[item.sector] || 0) + 1;
    });

    const sortedSectors = Object.entries(sectorCounts).sort((a, b) => b[1] - a[1]);
    const listContainer = document.getElementById('sectorsDensityList');
    listContainer.innerHTML = '';

    sortedSectors.forEach(([sector, count]) => {
        listContainer.innerHTML += `
            <li>
                <span class="sector-name">${sector}</span>
                <span class="sector-count-badge">${count} reportes</span>
            </li>
        `;
    });
}

// Pintar la línea de tiempo de auditoría transaccional (US04)
function renderAuditTimeline(logs) {
    const timeline = document.getElementById('auditTimeline');
    timeline.innerHTML = '';

    // 🚀 BLINDAJE 1: Controlar si la variable viene vacía o nula de forma segura
    if (!logs || logs.length === 0) {
        timeline.innerHTML = '<p class="empty-text">No hay movimientos registrados en la bitácora.</p>';
        return;
    }

    logs.forEach(log => {
        // 🚀 BLINDAJE 2: Gestión adaptativa del tiempo para SQLite
        // Si no existe la columna cambiado_en, usamos una etiqueta estática elegante con el ID del log
        let fechaFormateada = "Registro Guardado";
        if (log.cambiado_en) {
            fechaFormateada = new Date(log.cambiado_en).toLocaleString('es-BO');
        } else {
            fechaFormateada = `Operación Transaccional (Log #${log.id})`;
        }

        timeline.innerHTML += `
            <div class="timeline-item">
                <div class="timeline-marker"></div>
                <div class="timeline-content">
                    <span class="time-stamp"><i class="ti ti-clock"></i> ${fechaFormateada}</span>
                    <p><strong>${log.administrador_nombre || 'Administrador'}</strong> modificó la incidencia de <em>${log.tipo_danio || 'Daño Urbano'}</em></p>
                    <p class="transition-flow">
                        <span class="state-prev">${log.estado_anterior}</span> 
                        <i class="ti ti-arrow-narrow-right"></i> 
                        <span class="state-next">${log.estado_nuevo}</span>
                    </p>
                </div>
            </div>
        `;
    });
}

        method: 'PUT',
        headers: { 'Content-Type': 'application/json', 'Authorization': token },

    });
    
    const data = await res.json();
    if(data.status === 'success') {

    }
}

document.addEventListener("DOMContentLoaded", function() {
    const searchInput = document.getElementById('searchFilterInput');
    const sectorSelect = document.getElementById('sectorFilterSelect');
    const estadoSelect = document.getElementById('estadoFilterSelect');

    function ejecutarFiltrosCombinados() {
        const textoBusqueda = searchInput.value.toLowerCase().trim();
        const sectorSeleccionado = sectorSelect.value;
        const estadoSeleccionado = estadoSelect.value;

        // Aplicamos un súper filtro cruzado de tres niveles sobre la caché en memoria
        const incidenciasFiltradas = BACKEND_INCIDENCIAS_CACHE.filter(item => {
            
            // 1. Filtro por Texto (busca coincidencias en tipo de daño o descripción)
            const coincideTexto = item.tipo_danio.toLowerCase().includes(textoBusqueda) || 
                                 item.descripcion.toLowerCase().includes(textoBusqueda);
            
            // 2. Filtro por Sector (búsqueda parcial inteligente)
            const coincideSector = (sectorSeleccionado === "TODOS") || 
                                   (item.sector && item.sector.toLowerCase().includes(sectorSeleccionado.toLowerCase()));
            
            // 3. Filtro por Estado 
            const coincideEstado = (estadoSeleccionado === "TODOS") || 
                                   (item.estado === estadoSeleccionado);

            // La incidencia pasa solo si cumple las 3 condiciones al mismo tiempo
            return coincideTexto && coincideSector && coincideEstado;
        });

        // Volvemos a renderizar las tarjetas y a recalcular los contadores con el subconjunto filtrado
        renderMetricsAndIncidencias(incidenciasFiltradas);
    }

    // Escuchadores de eventos en tiempo real (Input para el teclado, Change para los selectores)
    searchInput.addEventListener('input', ejecutarFiltrosCombinados);
    sectorSelect.addEventListener('change', ejecutarFiltrosCombinados);
    estadoSelect.addEventListener('change', ejecutarFiltrosCombinados);
});
